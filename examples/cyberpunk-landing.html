// Drop-in config

One file. Zero loop changes.

# relay.yaml — start with defaults, tune as you grow
session:
  budget: $5.00
  resume: true

routing:
  default: deepseek-v4
  fallback: [ deepseek-v3, gpt-oss ]
  cache: read

tools:
  timeout: 30s
  retry: 2
  degrade: true
